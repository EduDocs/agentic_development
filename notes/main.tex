% Fixed 80-column frame width, measured in the actual listings font/size (so
% it stays correct if either changes) rather than stretching to \linewidth —
% a short one-line snippet no longer sits in a box sized for a paragraph.
\newlength{\lstlinewidth}

\lstset{
  % \upshape: without it, a listing inside an amsthm "plain"-style block
  % (Example, Theorem, ...) inherits that environment's italic body shape,
  % rendering the code slanted — and distorting the quote glyphs further.
  basicstyle=\ttfamily\upshape\small,
  breaklines=true,
  % listings' default columns=fixed forces every glyph into an equal-width
  % cell independent of the font's real metrics, which is what produces the
  % oddly letter-spaced look ("g i t   i n i t"); fullflexible uses the
  % font's own widths instead. keepspaces preserves the deliberate multiple
  % spaces used to align trailing "# comment" annotations.
  columns=fullflexible,
  keepspaces=true,
  % the upquote package only reaches listings when this option is set;
  % without it, straight quotes in code render as curly typographic quotes.
  upquote=true,
  showstringspaces=false,
  tabsize=4,
  breakautoindent=true,
  frame=single,
  linewidth=\lstlinewidth,
  xleftmargin=0pt,
  framexleftmargin=5pt,
  framexrightmargin=5pt,
  framextopmargin=3pt,
  framexbottommargin=3pt,
  backgroundcolor=\color{gray!8},
  keywordstyle=\color{blue!70!black},
  commentstyle=\color{gray!70},
  stringstyle=\color{green!55!black},
  numbers=left,
  numberstyle=\tiny\color{gray},
  numbersep=5pt,
  % listings' tokenizer is not UTF-8-aware by default; teach it the
  % typographic punctuation that appears in code comments (em/en dash,
  % curly quotes) rather than requiring every listing to avoid them.
  literate=
    {—}{{-{}-{}-}}1
    {–}{{-{}-}}1
    {’}{{\textquoteright}}1
    {‘}{{\textquoteleft}}1
    {”}{{\textquotedblright}}1
    {“}{{\textquotedblleft}}1,
}
