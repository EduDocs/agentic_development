%% leakage.tex: the sanctioned data flow of an evaluation harness and the one
%% forbidden path (Chapter 7). House conceptual style, plus a dashed red `leak'
%% arrow for the influence of held-out data on how a candidate is built or selected.
\input{figures/_concept-style}
\begin{tikzpicture}[
    box/.style={conceptbox, minimum width=2.7cm},
    arr/.style={conceptflow},
    leak/.style={->, >=stealth', thick, dashed, red}
]
\node[box, fill=gray!15]   (data)  at (0,0)      {dataset $\mathcal{D}$};
\node[box, fill=blue!15]    (train) at (-4.3,-1.9) {$\mathcal{D}_{\mathrm{train}}$};
\node[box, fill=yellow!35]  (val)   at (0,-1.9)    {$\mathcal{D}_{\mathrm{val}}$ (fitness)};
\node[box, fill=orange!25]  (test)  at (4.3,-1.9)  {$\mathcal{D}_{\mathrm{test}}$ (sealed)};
\node[box, fill=green!20]   (agent) at (-4.3,-4.0) {agent builds\\candidate $a$};
\node[box, fill=gray!15]    (fit)   at (0,-4.0)    {$\Score(a, \mathcal{D}_{\mathrm{val}})$};
\node[box, fill=gray!15]    (final) at (4.3,-4.0)  {$\Score(a^\star, \mathcal{D}_{\mathrm{test}})$};
\draw[arr] (data) -- node[above left, font=\scriptsize]{split first} (train);
\draw[arr] (data) -- (val);
\draw[arr] (data) -- (test);
\draw[arr] (train) -- (agent);
\draw[arr] (val)  -- (fit);
\draw[arr] (test) -- node[right, font=\scriptsize]{one-shot} (final);
\draw[arr] (agent) -- node[above, font=\scriptsize]{$a$} (fit);
\draw[arr] (fit) to[bend left=45] node[below, font=\scriptsize]{queried many times} (agent);
\draw[arr] (fit) -- node[above, font=\scriptsize]{winner $a^\star$} (final);
\draw[leak] (test.east) to[out=-45, in=-60, looseness=1.25]
                      node[midway, font=\scriptsize, below]{leakage (forbidden)}
                      node[pos=0.5, font=\large]{$\times$} (agent.south);
\end{tikzpicture}%
