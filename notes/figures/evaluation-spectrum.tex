%% evaluation-spectrum.tex — the evaluability spectrum (Chapter 7). House conceptual style.
%% Relational: one candidate, two scoring mechanisms. A verifiable ORACLE (top:
%% formula/test) and a judgment CRITIC (bottom: a capable model + rubric) both map
%% the same candidate to a score; a dashed "no formula" arrow shows the top lane
%% collapsing into the bottom one where no oracle exists (the critic IS the evaluator).
\input{figures/concept-style}
\begin{tikzpicture}[
    collapse/.style={->, >=stealth', thick, densely dashed, gray},
]
% shared input
\node[conceptbox, fill=yellow!20] (cand) at (0,0) {Candidate};

% the two scoring mechanisms
\node[conceptbox, fill=gray!15,   minimum width=3.6cm] (orac) at (4.6, 1.7) {Oracle\\{\scriptsize formula $\cdot$ test $\cdot$ check}};
\node[conceptbox, fill=violet!15, minimum width=3.6cm] (crit) at (4.6,-1.7) {Critic\\{\scriptsize capable model $+$ rubric}};

% the two scores
\node[conceptbox, fill=green!20] (so) at (9.0, 1.7) {score};
\node[conceptbox, fill=green!20] (sc) at (9.0,-1.7) {score};

% candidate branches to both mechanisms; each yields a score
\draw[conceptflow] (cand.north) |- (orac.west);
\draw[conceptflow] (cand.south) |- (crit.west);
\draw[conceptflow] (orac) -- (so);
\draw[conceptflow] (crit) -- (sc);

% the convergence, shown structurally: no oracle -> the critic scores instead
\draw[collapse] (orac.south) -- node[conceptlabel, right, text=black!55] {no formula} (crit.north);

% what each end is: pole word, properties, examples — placed OUTSIDE the lanes
\node[conceptlabel, anchor=south, text width=5cm] at (4.6, 2.55)
  {\textbf{verifiable}: objective, cheap, unambiguous\\ {\scriptsize sphere packing $\cdot$ code $\cdot$ MSE}};
\node[conceptlabel, anchor=north, text width=5cm] at (4.6,-2.55)
  {\textbf{judgment}: contextual, biased, gameable\\ {\scriptsize proof $\cdot$ prose $\cdot$ an idea}};
\end{tikzpicture}
